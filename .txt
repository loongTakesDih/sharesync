rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() {
      return request.auth != null;
    }

    function isRequestParticipant() {
      return signedIn() &&
        (
          resource.data.ownerId == request.auth.uid ||
          resource.data.borrowerId == request.auth.uid
        );
    }

    function requestExists(requestId) {
      return exists(/databases/$(database)/documents/requests/$(requestId));
    }

    function requestDoc(requestId) {
      return get(/databases/$(database)/documents/requests/$(requestId));
    }

    function isRequestParticipantById(requestId) {
      return signedIn() &&
        requestExists(requestId) &&
        (
          requestDoc(requestId).data.ownerId == request.auth.uid ||
          requestDoc(requestId).data.borrowerId == request.auth.uid
        );
    }

    function isValidChatMessage(requestId) {
      return request.resource.data.keys().hasAll([
          "text",
          "senderId",
          "senderUsername",
          "senderRole",
          "createdAt"
        ])
        && request.resource.data.keys().hasOnly([
          "text",
          "senderId",
          "senderUsername",
          "senderRole",
          "createdAt"
        ])
        && request.resource.data.text is string
        && request.resource.data.text.size() > 0
        && request.resource.data.text.size() <= 1000
        && request.resource.data.senderId == request.auth.uid
        && request.resource.data.senderUsername is string
        && request.resource.data.senderUsername.size() > 0
        && request.resource.data.senderUsername.size() <= 80
        && request.resource.data.createdAt == request.time
        && (
          (
            request.resource.data.senderRole == "owner" &&
            requestDoc(requestId).data.ownerId == request.auth.uid
          ) ||
          (
            request.resource.data.senderRole == "borrower" &&
            requestDoc(requestId).data.borrowerId == request.auth.uid
          )
        );
    }

    function isCompletedBorrowReview(requestId, itemId) {
      return signedIn()
        && requestExists(requestId)
        && requestDoc(requestId).data.itemId == itemId
        && requestDoc(requestId).data.borrowerId == request.auth.uid
        && requestDoc(requestId).data.status == "completed";
    }

    function isValidItemReviewCreate(requestId, itemId) {
      return isCompletedBorrowReview(requestId, itemId)
        && request.resource.data.keys().hasAll([
          "requestId",
          "itemId",
          "ownerId",
          "borrowerId",
          "borrowerUsername",
          "rating",
          "createdAt"
        ])
        && request.resource.data.keys().hasOnly([
          "requestId",
          "itemId",
          "ownerId",
          "borrowerId",
          "borrowerUsername",
          "borrowerProfileImageBase64",
          "rating",
          "comment",
          "createdAt",
          "updatedAt"
        ])
        && request.resource.data.requestId == requestId
        && request.resource.data.itemId == itemId
        && request.resource.data.ownerId == requestDoc(requestId).data.ownerId
        && request.resource.data.borrowerId == request.auth.uid
        && request.resource.data.borrowerId == requestDoc(requestId).data.borrowerId
        && request.resource.data.borrowerUsername is string
        && request.resource.data.borrowerUsername.size() > 0
        && request.resource.data.borrowerUsername.size() <= 80
        && request.resource.data.rating is number
        && request.resource.data.rating >= 1
        && request.resource.data.rating <= 5
        && request.resource.data.createdAt == request.time
        && (
          !request.resource.data.keys().hasAny(["updatedAt"]) ||
          request.resource.data.updatedAt == request.time
        )
        && (
          !request.resource.data.keys().hasAny(["comment"]) ||
          (
            request.resource.data.comment is string &&
            request.resource.data.comment.size() <= 300
          )
        )
        && (
          !request.resource.data.keys().hasAny(["borrowerProfileImageBase64"]) ||
          (
            request.resource.data.borrowerProfileImageBase64 is string &&
            request.resource.data.borrowerProfileImageBase64.size() <= 700000
          )
        );
    }

    function isValidItemReviewUpdate() {
      return signedIn()
        && resource.data.borrowerId == request.auth.uid
        && request.resource.data.diff(resource.data).affectedKeys()
          .hasOnly([
            "rating",
            "comment",
            "updatedAt"
          ])
        && request.resource.data.rating is number
        && request.resource.data.rating >= 1
        && request.resource.data.rating <= 5
        && request.resource.data.updatedAt == request.time
        && (
          !request.resource.data.keys().hasAny(["comment"]) ||
          (
            request.resource.data.comment is string &&
            request.resource.data.comment.size() <= 300
          )
        );
    }

    // USERS
    match /users/{userId} {
      allow read: if true;

      allow create: if signedIn() && request.auth.uid == userId;

      allow update: if signedIn() && (
        request.auth.uid == userId ||
        request.resource.data.diff(resource.data).affectedKeys()
          .hasOnly([
            "reputationScore",
            "trustRating",
            "trustStats",
            "trustScoreProcessed",
            "trustRatingProcessed",
            "trustRatingDelta"
          ])
      );

      allow delete: if signedIn() && request.auth.uid == userId;
    }

    // SAVED ITEMS
    // Path:
    // users/{uid}/savedItems/{itemId}
    match /users/{userId}/savedItems/{itemId} {
      allow read: if signedIn() && request.auth.uid == userId;

      allow create: if signedIn()
        && request.auth.uid == userId
        && request.resource.data.keys().hasAll([
          "itemId",
          "savedAt"
        ])
        && request.resource.data.keys().hasOnly([
          "itemId",
          "savedAt",
          "itemName",
          "itemImage",
          "ownerId",
          "communityId",
          "communityName"
        ])
        && request.resource.data.itemId == itemId
        && request.resource.data.savedAt == request.time;

      allow delete: if signedIn() && request.auth.uid == userId;

      allow update: if false;
    }

    // COMMUNITIES
    match /communities/{communityId} {
      allow read: if true;

      allow create: if signedIn()
        && request.resource.data.createdBy == request.auth.uid;

      allow update: if signedIn()
        && resource.data.createdBy == request.auth.uid
        && request.resource.data.diff(resource.data).affectedKeys()
          .hasOnly([
            "name",
            "nameLower",
            "area",
            "state",
            "country",
            "type",
            "active",
            "location",
            "requiresInvite",
            "inviteCode",
            "source",
            "externalId",
            "displayName",
            "updatedAt"
          ]);

      allow delete: if false;
    }

    // ITEMS
    match /items/{itemId} {
      allow read, write: if signedIn();

      // ITEM REVIEWS
      // Path:
      // items/{itemId}/reviews/{requestId}
      match /reviews/{requestId} {
        allow read: if signedIn();

        allow create: if isValidItemReviewCreate(requestId, itemId);

        allow update: if isValidItemReviewUpdate();

        allow delete: if false;
      }
    }

    // REQUESTS
    match /requests/{requestId} {
      allow create: if signedIn();

      allow read: if isRequestParticipant();

      allow update, delete: if isRequestParticipant();

      // REQUEST-BASED LIVE CHAT
      // Path:
      // requests/{requestId}/messages/{messageId}
      match /messages/{messageId} {
        allow read: if isRequestParticipantById(requestId);

        allow create: if isRequestParticipantById(requestId)
          && isValidChatMessage(requestId);

        allow update, delete: if false;
      }
    }
  }
}
